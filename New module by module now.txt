🧠 SERVER SCRIPT 1: Auto Generate Items

💻 Script
import frappe

def generate_items(doc):
   for row in doc.block_sizes:
       item_code = f"FB-{doc.density_code}-{int(row.length)}x{int(row.width)}x{int(row.height)}"
      
       if frappe.db.exists("Item", item_code):
           continue

       volume_m3 = (row.length * row.width * row.height) / 1000000000
       weight_kg = volume_m3 * doc.density_value

       item = frappe.get_doc({
           "doctype": "Item",
           "item_code": item_code,
           "item_name": item_code,
           "item_group": "Foam Blocks",
           "stock_uom": "Nos",
           "is_stock_item": 1,
           "custom_density": doc.density_value,
           "custom_length": row.length,
           "custom_width": row.width,
           "custom_height": row.height,
           "custom_weight": weight_kg,
           "custom_cbm": volume_m3
       })
       item.insert(ignore_permissions=True)

def on_update(doc, method):
   if doc.auto_generate_items and not doc.items_generated:
       generate_items(doc)
       doc.db_set("items_generated", 1)

🔷 2.3 Core Logic

🔹 2.3.1 Expected CBM
expected_cbm = runtime_minutes * output_rate

🔹 2.3.2 Actual CBM
total_cbm = qty * cbm_per_unit
actual_cbm = sum(total_cbm)

🔹 2.3.3 Variance
variance_cbm = actual_cbm - expected_cbm
variance_percent = (variance_cbm / expected_cbm) * 100

🔹 2.3.4 Cost Calculation
From Density Config → Operations Table:
cost_per_minute = hourly_cost / 60
total_cost = runtime_minutes * cost_per_minute

🔹 2.3.5 Batch Totals
total_expected_cbm = sum(expected_cbm)
total_actual_cbm = sum(actual_cbm)
total_variance_cbm = total_actual_cbm - total_expected_cbm
total_variance_percent = (total_variance_cbm / total_expected_cbm) * 100
total_production_cost = sum(total_cost)

🧠 STEP 3: Create Stock Entry (Transfer)
Type:
👉 Material Transfer

Script
def create_transfer_entry(doc):
   se = frappe.get_doc({
       "doctype": "Stock Entry",
       "stock_entry_type": "Material Transfer",
       "items": []
   })

   for row in doc.items:
       se.append("items", {
           "item_code": row.item_code,
           "qty": row.qty,
           "s_warehouse": doc.from_warehouse,
           "t_warehouse": doc.to_warehouse
       })

   se.insert(ignore_permissions=True)
   se.submit()

   return se.name

🧠 STEP 4: Status Progression
def on_submit(doc, method):
   if doc.status == "Approved":
       doc.status = "In Transit"
       doc.db_update()

       doc.stock_entry = create_transfer_entry(doc)

       doc.status = "Completed"
       doc.db_update()
